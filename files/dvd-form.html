DVD購入希望フォームを作るGAS

script.google.com の新しいプロジェクトに全部貼り付けて「実行」すると、Googleフォームと回答用スプレッドシートができます。 ← 作品集にもどる

/**
 * ファイナルコンサート DVD・ブルーレイ購入希望フォームを作る
 *
 * 使い方：
 *  1. https://script.google.com/ で「新しいプロジェクト」を開く
 *  2. この中身を全部貼り付けて保存
 *  3. 関数 createDvdForm を選んで「実行」→ 権限を許可
 *  4. 「実行ログ」に出る 回答用URL を、張り紙のQRに使う
 *     （フォームと回答スプレッドシートはマイドライブにできます）
 */
function createDvdForm() {
  const form = FormApp.create('ファイナルコンサート DVD・ブルーレイ 購入希望');
  form.setDescription(
    '登別中学校吹奏楽部 ファイナルコンサート（2026年10月18日）の記録映像を販売します。\n' +
    '\n' +
    '■ 価格：1枚 2,000円（DVD・ブルーレイとも）\n' +
    '■ 受け取り：学校に取りに来る　または　郵送（レターパックライト）\n' +
    '\n' +
    '【お支払い】\n' +
    '購入希望の方は、代金を現金で【10月18日中】に顧問の三浦までお支払いください。'
  );
  form.setConfirmationMessage(
    'ご回答ありがとうございました。\n' +
    '代金は現金で【10月18日中】に顧問の三浦までお支払いください。'
  );
  form.setProgressBar(false);

  // 1ページ目：基本情報
  form.addTextItem().setTitle('氏名').setRequired(true);

  form.addTextItem()
    .setTitle('メールアドレス')
    .setHelpText('発送や受け渡しのご連絡に使います。')
    .setRequired(true)
    .setValidation(FormApp.createTextValidation()
      .requireTextIsEmail()
      .setHelpText('メールアドレスの形で入力してください。')
      .build());

  form.addGridItem()
    .setTitle('購入希望（枚数）')
    .setHelpText('DVD・ブルーレイそれぞれの枚数を選んでください。片方だけの場合は、もう片方を「0枚」にしてください。')
    .setRows(['DVD', 'ブルーレイ'])
    .setColumns(['0枚', '1枚', '2枚', '3枚', '4枚', '5枚'])
    .setRequired(true);

  const receive = form.addMultipleChoiceItem()
    .setTitle('受け取り方法')
    .setRequired(true);

  // 2ページ目：郵送の人だけ
  const mailPage = form.addPageBreakItem()
    .setTitle('郵送先')
    .setHelpText('レターパックライトでお送りします。');
  form.addTextItem().setTitle('郵便番号').setHelpText('例：123-4567').setRequired(true);
  form.addParagraphTextItem().setTitle('住所').setHelpText('都道府県からマンション名・部屋番号まで').setRequired(true);

  // 受け取り方法で行き先を分ける（学校受け取り → そのまま送信）
  receive.setChoices([
    receive.createChoice('学校に取りに来る', FormApp.PageNavigationType.SUBMIT),
    receive.createChoice('郵送', mailPage),
  ]);

  // 回答をスプレッドシートにまとめる
  const ss = SpreadsheetApp.create('ファイナルコンサート DVD・ブルーレイ 購入希望（回答）');
  form.setDestination(FormApp.DestinationType.SPREADSHEET, ss.getId());

  Logger.log('回答用URL（QRに使う）: ' + form.shortenFormUrl(form.getPublishedUrl()));
  Logger.log('編集用URL: ' + form.getEditUrl());
  Logger.log('回答スプレッドシート: ' + ss.getUrl());
}
