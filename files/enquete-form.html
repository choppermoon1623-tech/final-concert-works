アンケートフォームを作るGAS

script.google.com の新しいプロジェクトに全部貼り付けて「実行」すると、Googleフォームと回答用スプレッドシートができます。 ← 作品集にもどる

/**
 * 登別中学校 ファイナルコンサート（第32回定期演奏会）アンケート
 * Googleフォームを作るスクリプト
 *
 * 使い方
 *  1. https://script.google.com/ を開き「新しいプロジェクト」
 *  2. 最初からあるコードを全部消して、このファイルの中身を貼り付けて保存
 *  3. 上の関数選択で「createForm」を選んで「実行」
 *     （初回は権限の確認が出ます →「詳細」→「〜に移動」→「許可」）
 *  4. 下の「実行ログ」に出る「回答用URL（短縮）」を、アンケート用紙の index.html に貼る
 *
 * フォームと回答用スプレッドシートは、マイドライブの一番上にできます。
 */

const TITLE = '登別中学校 ファイナルコンサート（第32回定期演奏会）アンケート';

const SONGS_1 = [
  'いつか王子様が（2・3年生6名のアンサンブル）',
  'ピースサイン',
  '糸',
  '情熱大陸',
  'ハンガリーの歌と民謡による組曲「子供のために」',
];

const SONGS_2 = [
  '愛するデューク',
  'Paradise Has No Border（指揮：渡會 大輔 先生）',
  'ディスコ・パーティー（指揮：佐土 登志男 先生）',
  '未来予想図Ⅱ（指揮：小林 聡子 先生）',
  '詩的なアリア（ホルン独奏：伊勢 久視 さん）',
  'オーメンズ・オブ・ラブ',
];

function createForm() {
  const form = FormApp.create(TITLE);
  form.setDescription(
    '登別中学校として最後の定期演奏会にご来場いただき、ありがとうございました。\n' +
    '32回の歴史を締めくくるこの日に、皆さまのお声をお聞かせください。いただいた言葉は、部員一人ひとりに届けます。\n\n' +
    '※ いただいたメッセージは、お名前を伏せて部員やOB・OGに紹介させていただくことがあります。'
  );
  form.setCollectEmail(false);
  form.setLimitOneResponsePerUser(false); // ログインなしで答えられるように
  form.setProgressBar(false);
  form.setConfirmationMessage('ご回答ありがとうございました。32年分の感謝を込めて。登別中学校吹奏楽部');

  // ① 心に残った曲
  form.addSectionHeaderItem()
    .setTitle('① 心に残った曲はどれですか？')
    .setHelpText('いくつでも選んでください。');
  form.addCheckboxItem()
    .setTitle('【第1部】現役生ステージ')
    .setChoiceValues(SONGS_1);
  form.addCheckboxItem()
    .setTitle('【第2部】OB・OGとの共演ステージ')
    .setChoiceValues(SONGS_2);

  // ② 本校との関わり
  form.addMultipleChoiceItem()
    .setTitle('② 本校との関わりを教えてください。')
    .setChoiceValues([
      '在校生',
      '保護者',
      '卒業生（吹奏楽部OB・OG）',
      '卒業生（吹奏楽部以外）',
      '地域の方（校区）',
      '地域の方（校区外）',
      '教職員・元教職員',
    ])
    .showOtherOption(true);

  // ③ 来場回数
  form.addMultipleChoiceItem()
    .setTitle('③ 登別中の定期演奏会に来られたのは何回目ですか？')
    .setChoiceValues(['今回がはじめて', '2〜4回目', '5回以上', '数えきれないほど']);

  // ④ 知ったきっかけ
  form.addCheckboxItem()
    .setTitle('④ 今日のコンサートをどのように知りましたか？')
    .setHelpText('いくつでも選んでください。')
    .setChoiceValues(['学校からの案内', 'ポスター', '知り合いから', 'ホームページ・SNS'])
    .showOtherOption(true);

  // ⑤ メッセージ
  form.addParagraphTextItem()
    .setTitle('⑤ 部員・OB・OG・先生方へのメッセージをお願いします。')
    .setHelpText('今日の演奏の感想や、これまでの登別中吹奏楽部の思い出も、ぜひお聞かせください。');

  // 回答用スプレッドシート
  const ss = SpreadsheetApp.create(TITLE + '（回答）');
  form.setDestination(FormApp.DestinationType.SPREADSHEET, ss.getId());

  const url = form.getPublishedUrl();
  let shortUrl = url;
  try { shortUrl = form.shortenFormUrl(url); } catch (e) {}

  Logger.log('回答用URL（短縮）: ' + shortUrl);
  Logger.log('回答用URL: ' + url);
  Logger.log('フォームの編集画面: ' + form.getEditUrl());
  Logger.log('回答のスプレッドシート: ' + ss.getUrl());
}
